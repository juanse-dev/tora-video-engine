> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One pure function, `compileAudioPlan`, decides every sound of a Story (where it starts, how long it plays, its gain on every frame); a thin React component turns the plan into `@remotion/media` `<Audio>` inside `<Sequence>`s; the Player starts muted, mounts the tracks only after muting itself, and shows volume controls only for Stories with audio. Stories without audio behave exactly as on `main`.

**Architecture:** Small pure modules (`ducking.ts`, `clipping.ts`, `musicLoop.ts`) compose into `plan.ts`; `sources.ts` holds the audio source contract (separate from the image one); `mounts.ts` turns a plan plus sources into the list of elements `StoryAudio.tsx` mounts, so the arming rule is unit-testable in Node; `ToraVideo` compiles the timeline once and hands that same object to `StoryRenderer` and `StoryAudio`. `Preview` keeps an `armed` state flipped by a layout effect that calls `player.mute()` first (pure helper `audioArming.ts`), remembers the volume in `localStorage` (`playerVolume.ts`) and never renders a Story's `<Audio>` before the Player is muted.

**Tech Stack:** TypeScript (Node `--experimental-strip-types`), React 19 / Remotion 4.0.529, `@remotion/media` 4.0.529 (new), `mediabunny` 1.56.1 (new, pinned for AUDIO-003/005), `node:test`, Playwright (bundled Chromium).

`src/audio/ducking.ts`:

```ts
/** A half-open frame interval `[start, end)` where an sfx or a voice sounds. */
export type DuckWindow = {start: number; end: number};

/**
 * Sorts the windows and merges those whose gap is smaller than
 * `mergeGapFrames` (overlapping windows have a negative gap and always merge).
 */
export const mergeDuckWindows = (
  windows: readonly DuckWindow[],
  mergeGapFrames: number,
): DuckWindow[] => {
  const sorted = [...windows].sort(
    (a, b) => a.start - b.start || a.end - b.end,
  );
  const merged: DuckWindow[] = [];

  for (const window of sorted) {
    const last = merged[merged.length - 1];

    if (last !== undefined && window.start - last.end < mergeGapFrames) {
      last.end = Math.max(last.end, window.end);
    } else {
      merged.push({start: window.start, end: window.end});
    }
  }

  return merged;
};

/**
 * Music gain factor for ducking (D-6): 1 outside the windows, `duck` inside, a
 * linear ramp 1 -> duck over `[start - ramp, start)` and duck -> 1 over
 * `[end, end + ramp)`. Overlapping ramps take the minimum. `windows` must come
 * from `mergeDuckWindows`.
 */
export const createDuckGain = (
  windows: readonly DuckWindow[],
  options: {duck: number; rampFrames: number},
): ((frame: number) => number) => {
  const {duck, rampFrames} = options;

  if (duck >= 1 || windows.length === 0) {
    return () => 1;
  }

  const windowGain = (window: DuckWindow, frame: number): number => {
    if (frame >= window.start && frame < window.end) {
      return duck;
    }

    if (frame < window.start) {
      return frame >= window.start - rampFrames
        ? 1 - ((1 - duck) * (frame - (window.start - rampFrames))) / rampFrames
        : 1;
    }

    return frame < window.end + rampFrames
      ? duck + ((1 - duck) * (frame - window.end)) / rampFrames
      : 1;
  };

  return (frame) => {
    // First window whose trailing ramp has not ended yet (ends ascend).
    let low = 0;
    let high = windows.length;

    while (low < high) {
      const middle = (low + high) >> 1;

      if (windows[middle].end + rampFrames > frame) {
        high = middle;
      } else {
        low = middle + 1;
      }
    }

    let gain = 1;

    for (
      let index = low;
      index < windows.length && windows[index].start - rampFrames <= frame;
      index += 1
    ) {
      gain = Math.min(gain, windowGain(windows[index], frame));
    }

    return gain;
  };
};
```

`src/audio/clipping.ts`:

```ts
export type ClippingTrack = {
  fromFrame: number;
  durationInFrames: number;
  gainAt: (absoluteFrame: number) => number;
};

/**
 * The largest amplitude sum, over all frames, of every track that is mounted
 * on that frame (D-7). One pass over the frames where something sounds, with
 * a small active set, so a 108,000-frame Story costs one pass, not one per track.
 */
export const maxSimultaneousGain = (
  tracks: readonly ClippingTrack[],
): number => {
  const pending = [...tracks].sort((a, b) => a.fromFrame - b.fromFrame);
  let active: ClippingTrack[] = [];
  let next = 0;
  let frame = 0;
  let best = 0;

  while (next < pending.length || active.length > 0) {
    if (active.length === 0) {
      frame = Math.max(frame, pending[next].fromFrame);
    }

    while (next < pending.length && pending[next].fromFrame <= frame) {
      active.push(pending[next]);
      next += 1;
    }

    active = active.filter(
      (track) => frame < track.fromFrame + track.durationInFrames,
    );

    if (active.length > 0) {
      let sum = 0;

      for (const track of active) {
        sum += track.gainAt(frame);
      }

      best = Math.max(best, sum);
      frame += 1;
    }
  }

  return best;
};
```

```ts
import {
  MIN_LOOPING_MUSIC_SECONDS,
  MUSIC_CROSSFADE_SECONDS,
} from "./constants.ts";

export type MusicCopy = {
  /** Position in `copies`; the plan key is `music-${index}`. */
  index: number;
  fromFrame: number;
  durationInFrames: number;
};

export type MusicLoopPlan = {
  copies: MusicCopy[];
  /** Overlap `X` between consecutive copies; 0 when the copies abut or there is one. */
  crossfadeFrames: number;
  /** True when the music is shorter than the minimum looping length and the Story is longer than it. */
  tooShortToLoop: boolean;
};

/**
 * Splits the music into overlapping copies that cover the Story (D-4).
 * Every length is at least one frame; copy starts are `k * (D - X)`.
 */
export const planMusicLoop = (input: {
  musicSeconds: number;
  totalFrames: number;
  fps: number;
  /** Defaults to MUSIC_CROSSFADE_SECONDS; a parameter so tests can exercise the clamp. */
  crossfadeSeconds?: number;
}): MusicLoopPlan => {
  const {
    musicSeconds,
    totalFrames,
    fps,
    crossfadeSeconds = MUSIC_CROSSFADE_SECONDS,
  } = input;
  const musicFrames = Math.max(1, Math.round(musicSeconds * fps));

  if (musicFrames >= totalFrames) {
    return {
      copies: [{index: 0, fromFrame: 0, durationInFrames: totalFrames}],
      crossfadeFrames: 0,
      tooShortToLoop: false,
    };
  }

  if (musicSeconds < MIN_LOOPING_MUSIC_SECONDS) {
    return {
      copies: [{index: 0, fromFrame: 0, durationInFrames: musicFrames}],
      crossfadeFrames: 0,
      tooShortToLoop: true,
    };
  }

  const crossfadeFrames = Math.min(
    Math.round(crossfadeSeconds * fps),
    Math.floor(musicFrames / 2),
  );
  const stride = musicFrames - crossfadeFrames;
  const copies: MusicCopy[] = [];

  for (let start = 0; start < totalFrames; start += stride) {
    copies.push({
      index: copies.length,
      fromFrame: start,
      durationInFrames: Math.min(musicFrames, totalFrames - start),
    });
  }

  return {copies, crossfadeFrames, tooShortToLoop: false};
};

/**
 * Equal-power crossfade factor of one copy on one frame (D-4): the incoming
 * copy follows `sin(pi*u/2)` over its first `X` frames, the outgoing copy
 * `cos(pi*u/2)` over the `X` frames where the next copy starts, with
 * `u = (frame - overlapStart + 0.5) / X`. 1 everywhere else.
 */
export const crossfadeFactorAt = (
  loop: MusicLoopPlan,
  copyIndex: number,
  frame: number,
): number => {
  const {copies, crossfadeFrames: overlap} = loop;

  if (overlap === 0) {
    return 1;
  }

  let factor = 1;
  const copy = copies[copyIndex];

  if (
    copyIndex > 0 &&
    frame >= copy.fromFrame &&
    frame < copy.fromFrame + overlap
  ) {
    factor *= Math.sin(
      (Math.PI / 2) * ((frame - copy.fromFrame + 0.5) / overlap),
    );
  }

  const next = copies[copyIndex + 1];

  if (
    next !== undefined &&
    frame >= next.fromFrame &&
    frame < next.fromFrame + overlap
  ) {
    factor *= Math.cos(
      (Math.PI / 2) * ((frame - next.fromFrame + 0.5) / overlap),
    );
  }

  return factor;
};

export type ResolvedFades = {
  fadeInSeconds: number;
  fadeOutSeconds: number;
  /** True when `fadeIn + fadeOut` exceeded the Story and both were scaled down (D-5). */
  scaled: boolean;
};

export const resolveFades = (input: {
  fadeInSeconds: number;
  fadeOutSeconds: number;
  totalFrames: number;
  fps: number;
}): ResolvedFades => {
  const {fadeInSeconds, fadeOutSeconds, totalFrames, fps} = input;
  const storySeconds = totalFrames / fps;
  const sum = fadeInSeconds + fadeOutSeconds;

  if (sum > storySeconds) {
    const scale = storySeconds / sum;

    return {
      fadeInSeconds: fadeInSeconds * scale,
      fadeOutSeconds: fadeOutSeconds * scale,
      scaled: true,
    };
  }

  return {fadeInSeconds, fadeOutSeconds, scaled: false};
};

/** Linear fade-in from frame 0 and fade-out to `totalFrames`, in [0, 1]. */
export const fadeFactorAt = (input: {
  frame: number;
  totalFrames: number;
  fps: number;
  fadeInSeconds: number;
  fadeOutSeconds: number;
}): number => {
  const {frame, totalFrames, fps, fadeInSeconds, fadeOutSeconds} = input;
  const fadeInFrames = fadeInSeconds * fps;
  const fadeOutFrames = fadeOutSeconds * fps;
  const fadeIn =
    fadeInFrames > 0 ? Math.min(1, Math.max(0, frame / fadeInFrames)) : 1;
  const fadeOut =
    fadeOutFrames > 0
      ? Math.min(1, Math.max(0, (totalFrames - frame) / fadeOutFrames))
      : 1;

  return fadeIn * fadeOut;
};
```

```ts
import type {Story} from "../story/types.ts";
import type {Timeline} from "../story/timeline.ts";
import type {AudioDurationMap, AudioRef} from "./catalog.ts";
import {maxSimultaneousGain} from "./clipping.ts";
import {DEFAULT_AUDIO_VOLUME, MAX_AUDIO_STORY_SECONDS} from "./constants.ts";
import {storyHasAudio} from "./storyAudio.ts";

export type AudioRole = "music" | "sfx" | "voice";

export type AudioPlanTrack = {
  /** Stable: "music-0", "music-1", "sfx-3", "voice-3". */
  key: string;
  role: AudioRole;
  ref: AudioRef;
  /** null for music. */
  sceneIndex: number | null;
  /** Absolute composition frame. */
  fromFrame: number;
  /** At least 1. */
  durationInFrames: number;
  /** In [0, 1]; pure; defined for [fromFrame, fromFrame + durationInFrames). */
  gainAt: (absoluteFrame: number) => number;
};

export type AudioPlanIssue =
  | {
      kind: "missing-audio";
      ref: AudioRef;
      role: AudioRole;
      sceneIndex: number | null;
    }
  | {
      kind: "pending-audio";
      ref: AudioRef;
      role: AudioRole;
      sceneIndex: number | null;
    }
  | {kind: "voice-cut"; sceneIndex: number; seconds: number}
  | {kind: "fades-scaled"}
  | {kind: "music-too-short-to-loop"}
  | {kind: "story-too-long-for-audio"};

export type AudioPlan = {
  tracks: readonly AudioPlanTrack[];
  issues: readonly AudioPlanIssue[];
  maxSimultaneousGain: number;
  /** The music's single-source envelope `volume x fade x duckGain` (no crossfade factors), for display; null without music. */
  musicEnvelopeAt: ((absoluteFrame: number) => number) | null;
};

export type AudioSourceState = "ready" | "pending" | "missing";

const clamp01 = (value: number): number => Math.min(1, Math.max(0, value));

/** A sub-frame source (AINV-7 only requires > 0) still gets one frame. */
const frameLength = (seconds: number, fps: number): number =>
  Math.max(1, Math.round(seconds * fps));

const isUsableDuration = (seconds: number | undefined): seconds is number =>
  typeof seconds === "number" && Number.isFinite(seconds) && seconds > 0;

const emptyPlan = (issues: readonly AudioPlanIssue[] = []): AudioPlan => ({
  tracks: [],
  issues,
  maxSimultaneousGain: 0,
  musicEnvelopeAt: null,
});

export const compileAudioPlan = (input: {
  story: Story;
  timeline: Timeline;
  fps: number;
  audioDurations: AudioDurationMap;
  sourceState: (ref: AudioRef) => AudioSourceState;
}): AudioPlan => {
  const {story, timeline, fps, audioDurations, sourceState} = input;

  if (!storyHasAudio(story)) {
    return emptyPlan();
  }

  const totalFrames = timeline.durationInFrames;

  // Bounds the work (and, with music, the copy count) in every caller.
  if (totalFrames > MAX_AUDIO_STORY_SECONDS * fps) {
    return emptyPlan([{kind: "story-too-long-for-audio"}]);
  }

  const issues: AudioPlanIssue[] = [];

  /** The ref's duration when its source is ready and measured; otherwise records exactly one issue. */
  const readySeconds = (
    ref: AudioRef,
    role: AudioRole,
    sceneIndex: number | null,
  ): number | null => {
    const state = sourceState(ref);

    if (state === "missing") {
      issues.push({kind: "missing-audio", ref, role, sceneIndex});

      return null;
    }

    const seconds = audioDurations[ref];

    if (state === "pending" || !isUsableDuration(seconds)) {
      issues.push({kind: "pending-audio", ref, role, sceneIndex});

      return null;
    }

    return seconds;
  };

  const sceneTracks: AudioPlanTrack[] = [];

  timeline.scenes.forEach((scene, sceneIndex) => {
    if (scene.sfx !== undefined) {
      const seconds = readySeconds(scene.sfx.src, "sfx", sceneIndex);

      if (seconds !== null) {
        const volume = clamp01(scene.sfx.volume ?? DEFAULT_AUDIO_VOLUME);

        sceneTracks.push({
          key: `sfx-${sceneIndex}`,
          role: "sfx",
          ref: scene.sfx.src,
          sceneIndex,
          fromFrame: scene.from,
          durationInFrames: Math.min(
            frameLength(seconds, fps),
            totalFrames - scene.from,
          ),
          gainAt: () => volume,
        });
      }
    }

    if (scene.voice !== undefined) {
      const seconds = readySeconds(scene.voice.src, "voice", sceneIndex);

      if (seconds !== null) {
        const volume = clamp01(scene.voice.volume ?? DEFAULT_AUDIO_VOLUME);
        const fullFrames = frameLength(seconds, fps);

        if (scene.duration !== "auto" && fullFrames > scene.durationInFrames) {
          issues.push({
            kind: "voice-cut",
            sceneIndex,
            seconds: (fullFrames - scene.durationInFrames) / fps,
          });
        }

        sceneTracks.push({
          key: `voice-${sceneIndex}`,
          role: "voice",
          ref: scene.voice.src,
          sceneIndex,
          fromFrame: scene.from,
          durationInFrames: Math.min(fullFrames, scene.durationInFrames),
          gainAt: () => volume,
        });
      }
    }
  });

  return {
    tracks: sceneTracks,
    issues,
    maxSimultaneousGain: maxSimultaneousGain(sceneTracks),
    musicEnvelopeAt: null,
  };
};
```

```ts
import type {Story} from "../story/types.ts";
import type {Timeline} from "../story/timeline.ts";
import type {AudioDurationMap, AudioRef} from "./catalog.ts";
import {maxSimultaneousGain} from "./clipping.ts";
import {
  DEFAULT_AUDIO_VOLUME,
  DEFAULT_MUSIC_DUCK,
  DUCK_MERGE_GAP_SECONDS,
  DUCK_RAMP_SECONDS,
  MAX_AUDIO_STORY_SECONDS,
} from "./constants.ts";
import {
  createDuckGain,
  mergeDuckWindows,
  type DuckWindow,
} from "./ducking.ts";
import {
  crossfadeFactorAt,
  fadeFactorAt,
  planMusicLoop,
  resolveFades,
} from "./musicLoop.ts";
import {storyHasAudio} from "./storyAudio.ts";

export type AudioRole = "music" | "sfx" | "voice";

export type AudioPlanTrack = {
  /** Stable: "music-0", "music-1", "sfx-3", "voice-3". */
  key: string;
  role: AudioRole;
  ref: AudioRef;
  /** null for music. */
  sceneIndex: number | null;
  /** Absolute composition frame. */
  fromFrame: number;
  /** At least 1. */
  durationInFrames: number;
  /** In [0, 1]; pure; defined for [fromFrame, fromFrame + durationInFrames). */
  gainAt: (absoluteFrame: number) => number;
};

export type AudioPlanIssue =
  | {
      kind: "missing-audio";
      ref: AudioRef;
      role: AudioRole;
      sceneIndex: number | null;
    }
  | {
      kind: "pending-audio";
      ref: AudioRef;
      role: AudioRole;
      sceneIndex: number | null;
    }
  | {kind: "voice-cut"; sceneIndex: number; seconds: number}
  | {kind: "fades-scaled"}
  | {kind: "music-too-short-to-loop"}
  | {kind: "story-too-long-for-audio"};

export type AudioPlan = {
  tracks: readonly AudioPlanTrack[];
  issues: readonly AudioPlanIssue[];
  maxSimultaneousGain: number;
  /** The music's single-source envelope `volume x fade x duckGain` (no crossfade factors), for display; null without music. */
  musicEnvelopeAt: ((absoluteFrame: number) => number) | null;
};

export type AudioSourceState = "ready" | "pending" | "missing";

const clamp01 = (value: number): number => Math.min(1, Math.max(0, value));

const frameLength = (seconds: number, fps: number): number =>
  Math.max(1, Math.round(seconds * fps));

const isUsableDuration = (seconds: number | undefined): seconds is number =>
  typeof seconds === "number" && Number.isFinite(seconds) && seconds > 0;

const emptyPlan = (
  issues: readonly AudioPlanIssue[] = [],
): AudioPlan => ({
  tracks: [],
  issues,
  maxSimultaneousGain: 0,
  musicEnvelopeAt: null,
});

export const compileAudioPlan = (input: {
  story: Story;
  timeline: Timeline;
  fps: number;
  audioDurations: AudioDurationMap;
  sourceState: (ref: AudioRef) => AudioSourceState;
}): AudioPlan => {
  const {story, timeline, fps, audioDurations, sourceState} = input;

  if (!storyHasAudio(story)) {
    return emptyPlan();
  }

  const totalFrames = timeline.durationInFrames;

  // Bounds the number of music copies (at most 5,143 at 30 FPS) in every caller.
  if (totalFrames > MAX_AUDIO_STORY_SECONDS * fps) {
    return emptyPlan([{kind: "story-too-long-for-audio"}]);
  }

  const issues: AudioPlanIssue[] = [];

  /** The ref's duration when its source is ready and measured; otherwise records exactly one issue. */
  const readySeconds = (
    ref: AudioRef,
    role: AudioRole,
    sceneIndex: number | null,
  ): number | null => {
    const state = sourceState(ref);

    if (state === "missing") {
      issues.push({kind: "missing-audio", ref, role, sceneIndex});

      return null;
    }

    const seconds = audioDurations[ref];

    if (state === "pending" || !isUsableDuration(seconds)) {
      issues.push({kind: "pending-audio", ref, role, sceneIndex});

      return null;
    }

    return seconds;
  };

  const musicSeconds =
    story.music === undefined
      ? null
      : readySeconds(story.music.src, "music", null);
  const musicLoop =
    musicSeconds === null
      ? null
      : planMusicLoop({musicSeconds, totalFrames, fps});

  if (musicLoop?.tooShortToLoop === true) {
    issues.push({kind: "music-too-short-to-loop"});
  }

  const fades =
    story.music === undefined || musicLoop === null
      ? null
      : resolveFades({
          fadeInSeconds: story.music.fadeIn ?? 0,
          fadeOutSeconds: story.music.fadeOut ?? 0,
          totalFrames,
          fps,
        });

  if (fades?.scaled === true) {
    issues.push({kind: "fades-scaled"});
  }

  const sceneTracks: AudioPlanTrack[] = [];

  timeline.scenes.forEach((scene, sceneIndex) => {
    if (scene.sfx !== undefined) {
      const seconds = readySeconds(scene.sfx.src, "sfx", sceneIndex);

      if (seconds !== null) {
        const volume = clamp01(scene.sfx.volume ?? DEFAULT_AUDIO_VOLUME);

        sceneTracks.push({
          key: `sfx-${sceneIndex}`,
          role: "sfx",
          ref: scene.sfx.src,
          sceneIndex,
          fromFrame: scene.from,
          durationInFrames: Math.min(
            frameLength(seconds, fps),
            totalFrames - scene.from,
          ),
          gainAt: () => volume,
        });
      }
    }

    if (scene.voice !== undefined) {
      const seconds = readySeconds(scene.voice.src, "voice", sceneIndex);

      if (seconds !== null) {
        const volume = clamp01(scene.voice.volume ?? DEFAULT_AUDIO_VOLUME);
        const fullFrames = frameLength(seconds, fps);

        if (scene.duration !== "auto" && fullFrames > scene.durationInFrames) {
          issues.push({
            kind: "voice-cut",
            sceneIndex,
            seconds: (fullFrames - scene.durationInFrames) / fps,
          });
        }

        sceneTracks.push({
          key: `voice-${sceneIndex}`,
          role: "voice",
          ref: scene.voice.src,
          sceneIndex,
          fromFrame: scene.from,
          durationInFrames: Math.min(fullFrames, scene.durationInFrames),
          gainAt: () => volume,
        });
      }
    }
  });

  const musicTracks: AudioPlanTrack[] = [];
  let musicEnvelopeAt: AudioPlan["musicEnvelopeAt"] = null;

  if (story.music !== undefined && musicLoop !== null && fades !== null) {
    const windows: DuckWindow[] = sceneTracks.map((track) => ({
      start: track.fromFrame,
      end: track.fromFrame + track.durationInFrames,
    }));
    const duckGainAt = createDuckGain(
      mergeDuckWindows(
        windows,
        Math.round(DUCK_MERGE_GAP_SECONDS * fps),
      ),
      {
        duck: clamp01(story.music.duck ?? DEFAULT_MUSIC_DUCK),
        rampFrames: Math.round(DUCK_RAMP_SECONDS * fps),
      },
    );
    const volume = clamp01(story.music.volume ?? DEFAULT_AUDIO_VOLUME);
    const envelopeAt = (frame: number): number =>
      clamp01(
        volume *
          fadeFactorAt({
            frame,
            totalFrames,
            fps,
            fadeInSeconds: fades.fadeInSeconds,
            fadeOutSeconds: fades.fadeOutSeconds,
          }) *
          duckGainAt(frame),
      );

    musicEnvelopeAt = envelopeAt;

    for (const copy of musicLoop.copies) {
      musicTracks.push({
        key: `music-${copy.index}`,
        role: "music",
        ref: story.music.src,
        sceneIndex: null,
        fromFrame: copy.fromFrame,
        durationInFrames: copy.durationInFrames,
        gainAt: (frame) =>
          envelopeAt(frame) * crossfadeFactorAt(musicLoop, copy.index, frame),
      });
    }
  }

  const tracks = [...musicTracks, ...sceneTracks];

  return {
    tracks,
    issues,
    maxSimultaneousGain: maxSimultaneousGain(tracks),
    musicEnvelopeAt,
  };
};
```

### Task 6: The audio source contract

**Files:**
- Create: `src/audio/sources.ts`, `tests/audio-sources.test.mjs`, `tests/types/audio-sources.typecheck.ts`

**Interfaces:**
- Consumes: `LocalAssetSource` (`src/localAssets/sources.ts`, unchanged), `LocalAudioRef` (`src/localAssets/refs.ts`), `BundledAudio`, `AudioRef`, `bundledAudioCatalog`, `isBundledAudio` (`src/audio/catalog.ts`), `storyAudioRefs` (`src/audio/storyAudio.ts`), `compileAudioPlan` (Tasks 4 and 5, for the integration tests), `staticFile` from `remotion`.
- Produces:
  - `type AudioSource = LocalAssetSource | {kind: "unavailable"}`
  - `type AudioSourceMap = Readonly<Partial<Record<LocalAudioRef | BundledAudio, AudioSource>>>` (keys narrowed to audio refs; the spec writes `LocalAssetRef`, audio-only is stricter and still accepts everything a producer builds)
  - `type LocalAudioSourceMap = Readonly<Partial<Record<LocalAudioRef, LocalAssetSource>>>` (what a producer of local sources hands in; the image `LocalAssetSourceMap` is assignable to it)
  - `type BundledAudioPolicy = "explicit" | "static-fallback"`
  - `type AudioSourceResolution = {kind: "audio"; src: string} | {kind: "pending"; ref: AudioRef} | {kind: "missing"; ref: AudioRef}`
  - `resolveAudioSrc(ref: AudioRef, sources: AudioSourceMap | undefined, policy: BundledAudioPolicy): AudioSourceResolution`
  - `buildAudioSourceMap(story: Story, localSources: LocalAudioSourceMap | undefined, bundled: (id: BundledAudio) => AudioSource | undefined): AudioSourceMap`
  - `audioSourceState(resolution: AudioSourceResolution): "ready" | "pending" | "missing"` (the `sourceState` mapping for `compileAudioPlan`)

Resolution rules: an entry `url` -> `{kind: "audio", src: url}`; `static` -> `staticFile(path)`; `pending` -> pending; `unavailable` -> missing (an entry always wins, also over the static fallback); inherited (prototype) keys are not entries. No entry: a local ref is missing; a bundled id is pending under `"explicit"`, and under `"static-fallback"` resolves to `staticFile(entry.file)` when `file` is not null, else missing (the two bundled music ids have `file: null`). It never resolves to another sound.

```ts
import {staticFile} from "remotion";
import type {LocalAudioRef} from "../localAssets/refs.ts";
import type {LocalAssetSource} from "../localAssets/sources.ts";
import type {Story} from "../story/types.ts";
import {
  bundledAudioCatalog,
  isBundledAudio,
  type AudioRef,
  type BundledAudio,
} from "./catalog.ts";
import {storyAudioRefs} from "./storyAudio.ts";

/**
 * Where an audio ref's bytes can be loaded from (runtime only, never
 * persisted). `unavailable` is the terminal state of a ref that cannot be
 * played; the image `LocalAssetSource` has no such state and stays unchanged.
 */
export type AudioSource = LocalAssetSource | {kind: "unavailable"};

export type AudioSourceMap = Readonly<
  Partial<Record<LocalAudioRef | BundledAudio, AudioSource>>
>;

/** What a producer of local sources hands to `buildAudioSourceMap`. */
export type LocalAudioSourceMap = Readonly<
  Partial<Record<LocalAudioRef, LocalAssetSource>>
>;

/** "explicit": a bundled id without an entry is pending. "static-fallback": it resolves to staticFile(entry.file). */
export type BundledAudioPolicy = "explicit" | "static-fallback";

export type AudioSourceResolution =
  | {kind: "audio"; src: string}
  | {kind: "pending"; ref: AudioRef}
  | {kind: "missing"; ref: AudioRef};

const entryFor = (
  sources: AudioSourceMap | undefined,
  ref: AudioRef,
): AudioSource | undefined =>
  sources !== undefined && Object.prototype.hasOwnProperty.call(sources, ref)
    ? sources[ref]
    : undefined;

/**
 * Pure resolution of an audio ref to something playable. Never falls back to
 * another sound (AINV-5): a ref without a usable source is pending or missing.
 */
export const resolveAudioSrc = (
  ref: AudioRef,
  sources: AudioSourceMap | undefined,
  policy: BundledAudioPolicy,
): AudioSourceResolution => {
  const entry = entryFor(sources, ref);

  if (entry !== undefined) {
    switch (entry.kind) {
      case "url":
        return {kind: "audio", src: entry.url};
      case "static":
        return {kind: "audio", src: staticFile(entry.path)};
      case "pending":
        return {kind: "pending", ref};
      case "unavailable":
        return {kind: "missing", ref};
    }
  }

  if (!isBundledAudio(ref)) {
    return {kind: "missing", ref};
  }

  if (policy === "explicit") {
    return {kind: "pending", ref};
  }

  const {file} = bundledAudioCatalog[ref];

  return file === null
    ? {kind: "missing", ref}
    : {kind: "audio", src: staticFile(file)};
};

/**
 * Assembles the `audioSources` prop of a Story from the sources each producer
 * already owns: a `local:audio:` ref copies the same entry object from
 * `localSources` (no new object URL, no revocation: ownership, leases and
 * `pending` stay with the producer); a bundled id takes `bundled(id)`. Refs the
 * Story does not use are never included and a ref without an entry is absent.
 */
export const buildAudioSourceMap = (
  story: Story,
  localSources: LocalAudioSourceMap | undefined,
  bundled: (id: BundledAudio) => AudioSource | undefined,
): AudioSourceMap => {
  const map: Partial<Record<LocalAudioRef | BundledAudio, AudioSource>> = {};

  for (const ref of storyAudioRefs(story)) {
    const entry = isBundledAudio(ref)
      ? bundled(ref)
      : localSources !== undefined &&
          Object.prototype.hasOwnProperty.call(localSources, ref)
        ? localSources[ref]
        : undefined;

    if (entry !== undefined) {
      map[ref] = entry;
    }
  }

  return map;
};

/** Maps a resolution to the `sourceState` that `compileAudioPlan` expects. */
export const audioSourceState = (
  resolution: AudioSourceResolution,
): "ready" | "pending" | "missing" =>
  resolution.kind === "audio" ? "ready" : resolution.kind;
```

```ts
import type {AudioPlan} from "./plan.ts";
import {
  resolveAudioSrc,
  type AudioSourceMap,
  type BundledAudioPolicy,
} from "./sources.ts";

/** One `<Sequence>` + `<Audio>` pair that `StoryAudio` mounts. */
export type StoryAudioMount = {
  /** Plan key plus the source, so a changed source remounts the element. */
  key: string;
  from: number;
  durationInFrames: number;
  src: string;
  /** Remotion `volume` callback: the frame is relative to the sequence. */
  volumeAt: (frame: number) => number;
};

/**
 * The elements `StoryAudio` mounts for a plan. Mounts nothing only when
 * `audioArmed` is exactly `false`; absent means armed (Remotion Studio and any
 * composition rendered without a Player play audio).
 */
export const buildStoryAudioMounts = (input: {
  plan: AudioPlan;
  sources: AudioSourceMap | undefined;
  policy: BundledAudioPolicy;
  audioArmed: boolean | undefined;
}): StoryAudioMount[] => {
  const {plan, sources, policy, audioArmed} = input;

  if (audioArmed === false) {
    return [];
  }

  return plan.tracks.flatMap((track) => {
    const resolution = resolveAudioSrc(track.ref, sources, policy);

    return resolution.kind === "audio"
      ? [
          {
            key: `${track.key}:${resolution.src}`,
            from: track.fromFrame,
            durationInFrames: track.durationInFrames,
            src: resolution.src,
            volumeAt: (frame: number) => track.gainAt(track.fromFrame + frame),
          },
        ]
      : [];
  });
};
```

`src/web/audioArming.ts`:

```ts
export type ArmingAction = "arm" | "disarm" | "none";

/**
 * The tracks of a Story with audio are mounted only after the Player is
 * muted. "arm" is needed on the first render that has audio and after every
 * period without audio; editing a Story that stays audible changes nothing,
 * so a user who unmuted is not muted again.
 */
export const nextArmingAction = (
  hasAudio: boolean,
  armed: boolean,
): ArmingAction => {
  if (hasAudio) {
    return armed ? "none" : "arm";
  }

  return armed ? "disarm" : "none";
};

/**
 * Applies the action. Arming mutes the Player first and only then sets the
 * state that lets `StoryAudio` mount, so the first commit containing an
 * `<Audio>` happens after the Player is muted. Without a Player yet it does
 * nothing and returns false, so the caller tries again.
 */
export const applyArmingAction = (
  action: ArmingAction,
  player: {mute: () => void} | null,
  setArmed: (armed: boolean) => void,
): boolean => {
  if (action === "none") {
    return true;
  }

  if (action === "disarm") {
    setArmed(false);

    return true;
  }

  if (player === null) {
    return false;
  }

  player.mute();
  setArmed(true);

  return true;
};
```

```ts
export const PLAYER_VOLUME_STORAGE_KEY = "tora.playerVolume";

type VolumeStorage = Pick<Storage, "getItem" | "setItem">;

/** Returns null when storage is unavailable; reading `window.localStorage` itself can throw. */
const browserStorage = (): VolumeStorage | null => {
  try {
    return window.localStorage;
  } catch {
    return null;
  }
};

const isVolume = (value: number): boolean =>
  Number.isFinite(value) && value >= 0 && value <= 1;

/**
 * The remembered Player volume, or null when nothing usable is stored. A
 * convenience only: every failure (no storage, a throwing getter, garbage,
 * a number outside [0, 1]) yields null.
 */
export const readStoredPlayerVolume = (
  getStorage: () => VolumeStorage | null = browserStorage,
): number | null => {
  try {
    const raw = getStorage()?.getItem(PLAYER_VOLUME_STORAGE_KEY);

    if (typeof raw !== "string" || raw.trim() === "") {
      return null;
    }

    const value = Number(raw);

    return isVolume(value) ? value : null;
  } catch {
    return null;
  }
};

/** Remembers the volume; silently does nothing when storage fails or the value is not in [0, 1]. */
export const storePlayerVolume = (
  volume: number,
  getStorage: () => VolumeStorage | null = browserStorage,
): void => {
  try {
    if (!isVolume(volume)) {
      return;
    }

    getStorage()?.setItem(PLAYER_VOLUME_STORAGE_KEY, String(volume));
  } catch {
    // The remembered volume is a convenience, never required.
  }
};
```

```mjs
// Records every audio start and every mute change with a timestamp, and
// whether the Player showed itself as muted at that moment (the volume button
// reads "Unmute sound" while muted). Remotion's own silent shared <audio>
// pool (data: URLs) is flagged so it can be ignored.
const installAudioSpy = () => {
  window.__audioLog = [];
  window.__muteLog = [];

  const isMutedNow = () =>
    document.querySelector('.preview-frame button[aria-label="Unmute sound"]') !== null;

  for (const [kind, prototype] of [
    ["AudioBufferSourceNode.start", AudioBufferSourceNode.prototype],
    ["AudioScheduledSourceNode.start", AudioScheduledSourceNode.prototype],
  ]) {
    const original = prototype.start;

    prototype.start = function start(...args) {
      window.__audioLog.push({
        kind,
        t: performance.now(),
        muted: isMutedNow(),
        pool: false,
      });

      return original.apply(this, args);
    };
  }

  const originalPlay = HTMLMediaElement.prototype.play;

  HTMLMediaElement.prototype.play = function play(...args) {
    window.__audioLog.push({
      kind: "HTMLMediaElement.play",
      t: performance.now(),
      muted: isMutedNow(),
      pool: this.src.startsWith("data:"),
    });

    return originalPlay.apply(this, args);
  };

  new MutationObserver(() => {
    const value = document
      .querySelector(".preview-frame")
      ?.getAttribute("data-tora-muted");

    if (value !== undefined && value !== null) {
      const last = window.__muteLog[window.__muteLog.length - 1];

      if (last === undefined || last.value !== value) {
        window.__muteLog.push({t: performance.now(), value});
      }
    }
  }).observe(document, {attributes: true, subtree: true, childList: true});
};

};

  // Seeking: a click at three quarters of the seek bar lands near frame 224 of 300.
  const bar = await page

      const found = elements
        .map((element) => ({
          rect: element.getBoundingClientRect(),
          style: getComputedStyle(element),
        }))
        .find(
          ({rect, style}) =>
            style.cursor === "pointer" &&
            style.touchAction === "none" &&
            rect.width > 200,
        );

      return found === undefined
        ? null
        : {
            x: found.rect.x,
            y: found.rect.y + found.rect.height / 2,
            width: found.rect.width,
          };
    });

  expect(bar).not.toBeNull();
  await page.mouse.click(bar.x + bar.width * 0.75, bar.y);
  await expect
    .poll(() => currentFrame(page), {timeout: 5_000})

  page,
}) => {
```
